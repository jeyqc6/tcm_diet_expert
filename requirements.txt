# diet_expert · Python 依赖清单
# 换机子/新环境一键安装：pip install -r requirements.txt

# 各包对应哪个脚本、能不能跳过，见下面分组注释；
# 具体用法见各脚本文件头的 docstring。

# ---- 知识库摄入（planning/step1-naive-rag/ingest.py）----
pdfplumber          # 有文字层 PDF 逐页解析 + 表格抽取兜底路径
beautifulsoup4      # NIH XML 补剂说明按 h2/h3 解析
lxml                # BeautifulSoup 用 "lxml" 作为解析器
transformers        # AutoTokenizer.from_pretrained("BAAI/bge-m3")，token 感知二次切分用

# ---- 配置（backend/env.py — 从 .env 读 DIET_EXPERT_PG_DSN 等）----
python-dotenv

# ---- Embedding + 入库（db/embed_bge_m3.py, db/load_recipes.py）----
FlagEmbedding       # BGE-M3 dense embedding 模型
torch               # FlagEmbedding 的底层依赖
psycopg2-binary     # 连 Postgres
pgvector>=0.5.0     # Python 包：vector/sparsevec <-> numpy 互转；SparseVector 从 0.5.0 起有
                    # （PyPI 上目前最高 0.5.0，别和 Postgres 扩展版本号混了）
                    # Postgres 端扩展也需支持 sparsevec（docker-compose 用 pgvector/pgvector:pg16）
numpy

# ---- Step1 naive RAG 原型 / 评测（planning/step1-naive-rag/）----
scikit-learn        # build_and_eval_bm25.py 用 CountVectorizer 做 BM25 对比

# ---- LLM adapter（backend/llm/adapter.py + providers/）----
# 三选一或都装都行——用哪个由 .env 里的 LLM_PROVIDER_DEV/LLM_PROVIDER_PROD 决定，
# 不是装了就一定用，见 .env.example 的说明。
openai              # provider=openai 或 provider=ollama（Ollama 自带 OpenAI 兼容层，复用同一个包）
anthropic           # provider=anthropic
# Ollama 本身不是 pip 包——它是本机常驻服务，用 `brew install ollama`（macOS）装，
# 装完 `ollama pull <model>` 拉一个模型、`ollama serve` 常驻起来（一般装完会自动跑）

# ---- API 层（api/main.py，D26）----
fastapi
uvicorn             # 生产/本地都用它跑：uvicorn api.main:app
sse-starlette       # 目前 api/main.py 用的是手写 SSE(纯字符串拼 "event:.../data:...")，
                    # 没有直接依赖这个包；先声明是因为 FastAPI 生态标准的 SSE 方案就是它，
                    # 后续要换成官方 EventSourceResponse 时不用再补依赖
httpx               # fastapi.testclient.TestClient 需要，tests/integration/test_api_chat_sse.py 用

# ---- 测试（tests/ + .github/workflows/ci.yml）----
pytest

# ---- 可观测（ENGINEERING §6，阶段 6 Langfuse 全链路）----

langfuse>=3.0,<4

ruff
